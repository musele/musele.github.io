{{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $resource := or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- if $resource -}} {{- $src = $resource.RelPermalink -}} {{- else if hasPrefix $path "media/" -}} {{- /* Obsidian 的 media/ 附件目录位于文章所属的顶层分区下，例如 content/posts/media */ -}} {{- $src = printf "%s%s" .Page.FirstSection.RelPermalink $path -}} {{- end -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- $attributes := merge .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy") -}}
